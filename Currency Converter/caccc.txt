def normalize_key(key):
    if not key:
        return None
    k = str(key).strip().lower()
    k = re.sub(r"[^a-z0-9\s]", " ", k)
    k = re.sub(r"\s+", " ", k)
    return NORMALIZE_KEYS.get(k, None)

# ======================================================
# UTILITIES
# ======================================================
def find_excel_files(root):
    files = []
    for dp, _, fns in os.walk(root):
        for f in fns:
            if f.lower().endswith((".xlsx", ".xls")):
                files.append(os.path.join(dp, f))
    return files

def extract_story_id(sheet):
    for row in sheet.iter_rows(values_only=True):
        for val in row:
            if val and isinstance(val, str):
                m = re.search(STORY_REGEX, val, re.IGNORECASE)
                if m:
                    return m.group(0).upper()
    return None

def clean_value(val):
    if val is None:
        return None
    v = str(val).strip()
    return None if v.lower() in ("", "none", "nan", "null") else v

# ======================================================
# INGESTION
# ======================================================
def ingest_file(file_path, conn):
    try:
        wb = load_workbook(file_path, data_only=True)
        sheet = wb.active
        story_id = extract_story_id(sheet)
        if not story_id:
            info_logger.warning(f"No Story ID found in {file_path}")
            return

        metadata_dict = {"story_id": story_id}
        extra_fields = {}
        result_started = False
        result_rows = []

        for row in sheet.iter_rows(values_only=True):
            cells = [clean_value(c) for c in row]
            if not any(cells):
                continue

            first_cell = str(cells[0]).strip().lower() if cells[0] else ""

            if not result_started:
                normalized = normalize_key(first_cell)
                if normalized:
                    metadata_dict[normalized] = cells[1] if len(cells) > 1 else None
                    continue

                if len(cells) >= 2:
                    key = str(cells[0]).strip()
                    val = str(cells[1]).strip() if cells[1] else None
                    if key and val:
                        extra_fields[key] = val
                    continue

                if "results" in first_cell:
                    result_started = True
                    continue

            else:
                filtered_row = [c for c in cells if c not in (None, "", " ")]
                if filtered_row:
                    result_rows.append(filtered_row)

        if extra_fields:
            metadata_dict["extra_metadata"] = json.dumps(extra_fields, ensure_ascii=False)

        with conn.begin():
            conn.execute(story_metadata.insert().values(**metadata_dict))

        if result_rows:
            transposed = list(zip(*result_rows))
            non_empty_indices = [
                i for i, col_vals in enumerate(transposed)
                if any(v not in (None, "", " ") for v in col_vals)
            ]
            compact_rows = [[row[i] for i in non_empty_indices if i < len(row)] for row in result_rows]

            for row in compact_rows:
                record = {"story_id": story_id}
                for i, val in enumerate(row[:6]):
                    record[f"col{i+1}"] = val
                with conn.begin():
                    conn.execute(story_results.insert().values(**record))

        info_logger.info(f"{story_id}: {len(result_rows)} result rows from {os.path.basename(file_path)}")
        print(f"{story_id}: {len(result_rows)} result rows inserted")

    except SQLAlchemyError as e:
        error_logger.error(f"DB error for {file_path}: {e}", exc_info=True)
    except Exception as e:
        error_logger.error(f"Processing error for {file_path}: {e}", exc_info=True)

# ======================================================

if __name__ == "__main__":

        for f in files:
            print(f"Processing: {f}")
            ingest_file(f, conn)

    print("Ingestion complete. Check logs in /logs/")
