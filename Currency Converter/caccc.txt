# PER-FILE EXCEL LOGGING

def _excel_log_path(source_path: str) -> str:
    base = os.path.splitext(os.path.basename(source_path))[0]
    safe = re.sub(r"[^A-Za-z0-9_.-]+", "_", base)
    return os.path.join("file_logs", f"{safe}.xlsx")

def xlog_init(path: str):
    """Create a new Excel log file with headers if it doesn't exist."""
    if os.path.exists(path):
        return
    wb = Workbook()
    ws = wb.active
    ws.title = "Log"
    ws.append(["timestamp", "level", "message"])
    wb.save(path)
    wb.close()

def xlog_write(path: str, level: str, message: str):
    """Append a row to the Excel log."""
    try:
        if not os.path.exists(path):
            xlog_init(path)
        wb = load_workbook(path)
        ws = wb["Log"]
        ws.append([datetime.now().isoformat(timespec="seconds"), level.upper(), message])
        wb.save(path)
        wb.close()
    except Exception as e:
        # Fall back to master log if Excel write fails
        logger.error(f"Per-file Excel log failed for {path}: {e}", exc_info=True)

def flog(path: str, level: str, message: str):
    """Log both to master and per-file Excel log."""
    getattr(logger, level.lower())(f"[{os.path.basename(path)}] {message}")
    xlog_write(_excel_log_path(path), level, message)

def normalize_key(key):
    if not key:
        return None
    k = str(key).lower()
    k = re.sub(r"[^a-z0-9\s:/-]", " ", k)
    k = re.sub(r"\s+", " ", k).strip()
    return NORMALIZE_KEYS.get(k)

# CLEANERS / HELPERS
# ======================================================
ZERO_WIDTH = {"\u200b", "\u200c", "\u200d", "\ufeff"}
NBSP = "\xa0"

def _strip_specials(s: str) -> str:
    if not isinstance(s, str):
        s = str(s)
    # keep tabs/newlines for now, then collapse whitespace later
    s = "".join(ch for ch in s if unicodedata.category(ch)[0] != "C" or ch in "\t\n\r")
    for zw in ZERO_WIDTH:
        s = s.replace(zw, "")
    s = s.replace(NBSP, " ")
    # collapse whitespace including \r \n \t
    s = " ".join(s.split())
    return s.strip()

def clean_value(val):
    if val is None:
        return None
    s = _strip_specials(str(val))
    return None if s == "" or s.lower() in {"none", "nan", "null"} else s

def find_excel_files(root):
    files = []
    for dp, _, fns in os.walk(root):
        for f in fns:
            lf = f.lower()
            # openpyxl supports these; we exclude legacy .xls
            if lf.endswith((".xlsx", ".xlsm", ".xltx", ".xltm")):
                files.append(os.path.join(dp, f))
    return files

def extract_story_id(sheet):
    pat = re.compile(STORY_REGEX, re.IGNORECASE)
    for row in sheet.iter_rows(values_only=True):
        for val in row:
            if isinstance(val, str):
                m = pat.search(val)
                if m:
                    return m.group(0).upper()
    return None

# INGESTION
# ======================================================
def ingest_file(file_path: str, conn):
    xlog_init(_excel_log_path(file_path))
    flog(file_path, "info", "Starting ingestion")

    try:
        wb = load_workbook(file_path, data_only=True, read_only=True)
    except Exception as e:
        flog(file_path, "error", f"Failed to open workbook: {e}")
        return

    try:
        sheet = wb.active

        story_id = extract_story_id(sheet)
        if not story_id:
            flog(file_path, "warning", "No Story ID found. Skipping file.")
            return
        flog(file_path, "info", f"Detected story_id: {story_id}")

        metadata_dict = {"story_id": story_id}
        extra_fields = {}
        result_started = False
        result_rows = []

        for row in sheet.iter_rows(values_only=True):
            # Clean cells once
            cells = [clean_value(c) for c in row]
            # If all cleaned cells are None -> skip
            if not any(c is not None for c in cells):
                continue

            # Detect results header anywhere on the row (ignoring junk/newlines)
            if not result_started:
                if any((_strip_specials(c or "").lower() in RESULT_TOKENS) for c in cells if c is not None):
                    result_started = True
                    flog(file_path, "info", "Entered results section")
                    continue

                # Try "Key: Value" in any cell
                kv_matched = False
                for c in cells:
                    if c and ":" in c:
                        k, v = c.split(":", 1)
                        k, v = clean_value(k), clean_value(v)
                        if k and v:
                            norm = normalize_key(k)
                            if norm:
                                metadata_dict[norm] = v
                                flog(file_path, "info", f"Metadata[{norm}] = <set>")
                            else:
                                extra_fields[k] = v
                                flog(file_path, "info", f"ExtraMeta[{k}] = <set>")
                            kv_matched = True
                            break
                if kv_matched:
                    continue

                # Fallback: 2-column KV (A->key, B->val) if both non-empty
                if len(cells) >= 2 and cells[0] is not None and cells[1] is not None:
                    norm = normalize_key(cells[0])
                    if norm:
                        metadata_dict[norm] = cells[1]
                        flog(file_path, "info", f"Metadata[{norm}] = <set>")
                    else:
                        extra_fields[cells[0]] = cells[1]
                        flog(file_path, "info", f"ExtraMeta[{cells[0]}] = <set>")
                    continue

                # Not results, not metadata — ignore noisy row
                continue

            # RESULTS mode: pack non-empty cells leftwards per row
            non_empty = [c for c in cells if c is not None]
            if non_empty:
                result_rows.append(non_empty)

        # Attach extra fields JSON if present
        if extra_fields:
            metadata_dict["extra_metadata"] = json.dumps(extra_fields, ensure_ascii=False)

        # Insert in a single transaction
        try:
            with conn.begin():
                # Only insert metadata keys that actually exist (already ensured)
                conn.execute(story_metadata.insert().values(**metadata_dict))

                if result_rows:
                    records = []
                    for row_vals in result_rows:
                        rec = {"story_id": story_id}
                        for i, val in enumerate(row_vals[:MAX_RESULT_COLS]):
                            rec[f"col{i+1}"] = val
                        records.append(rec)
                    conn.execute(story_results.insert(), records)

            flog(file_path, "info", f"Inserted metadata + {len(result_rows)} result rows")
        except SQLAlchemyError as e:
            flog(file_path, "error", f"Database error: {e}")
            return

    except Exception as e:
        flog(file_path, "error", f"Processing error: {e}")
        return
    finally:
        try:
            wb.close()
        except Exception:
            pass

    flog(file_path, "info", "Ingestion finished")
